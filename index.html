Query SQL Unificada para BisExpress (Sem Comentários — Pronta para JSON ou Excel)

Une SC7 → SD1 → SF1 → SE2 → SE5 → SA2 → SY1 em uma única execução.

WITH CTE_SE5_BAIXAS AS (
    SELECT
        RTRIM(SE5.E5_FILIAL) AS Filial_Baixa,
        RTRIM(SE5.E5_PREFIXO) AS Prefixo_Baixa,
        RTRIM(SE5.E5_NUMERO) AS Numero_Titulo,
        RTRIM(SE5.E5_PARCELA) AS Parcela_Titulo,
        RTRIM(SE5.E5_TIPO) AS Tipo_Titulo,
        RTRIM(SE5.E5_CLIFOR) AS Cod_Fornecedor,
        RTRIM(SE5.E5_LOJA) AS Loja_Fornecedor,
        MAX(CASE WHEN ISDATE(SE5.E5_DATA) = 1 THEN CAST(SE5.E5_DATA AS DATE) END) AS DT_Movimentacao_SE5,
        MAX(CASE WHEN ISDATE(SE5.E5_DTDIGIT) = 1 THEN CAST(SE5.E5_DTDIGIT AS DATE) END) AS DT_Digitacao_SE5,
        SUM(SE5.E5_VALOR) AS Valor_Movimentado_SE5,
        MAX(RTRIM(SE5.E5_DOCUMEN)) AS Documento_SE5,
        MAX(RTRIM(SE5.E5_MOTBX)) AS Motivo_Baixa_SE5,
        MAX(RTRIM(SE5.E5_TIPODOC)) AS Tipo_Doc_SE5
    FROM SE5010 SE5 WITH (NOLOCK)
    WHERE SE5.D_E_L_E_T_ = ''
      AND SE5.E5_RECPAG = 'P'
      AND SE5.E5_TIPODOC <> 'PA'
      AND SE5.E5_MOTBX = 'DEB'
      AND SE5.E5_VALOR <> 0
      AND SE5.E5_SITUACA <> 'C'
      AND SE5.E5_DATA >= '20260101'
    GROUP BY
        RTRIM(SE5.E5_FILIAL),
        RTRIM(SE5.E5_PREFIXO),
        RTRIM(SE5.E5_NUMERO),
        RTRIM(SE5.E5_PARCELA),
        RTRIM(SE5.E5_TIPO),
        RTRIM(SE5.E5_CLIFOR),
        RTRIM(SE5.E5_LOJA)
),
CTE_PEDIDOS_NF AS (
    SELECT
        COALESCE(RTRIM(SD1.D1_FILIAL), RTRIM(SC7.C7_FILIAL)) AS Filial_Origem,
        RTRIM(SC7.C7_NUM) AS Numero_Pedido,
        RTRIM(SC7.C7_ITEM) AS Item_Pedido,
        CASE WHEN ISDATE(SC7.C7_EMISSAO) = 1 THEN CAST(SC7.C7_EMISSAO AS DATE) END AS Data_Emissao_Pedido,
        RTRIM(SC7.C7_COND) AS Condicao_Pagto_Pedido,
        RTRIM(SC7.C7_USER) AS Cod_Usuario_Comprador,
        RTRIM(SY1.Y1_NOME) AS Nome_Comprador,
        COALESCE(RTRIM(SD1.D1_COD), RTRIM(SC7.C7_PRODUTO)) AS Cod_Produto,
        RTRIM(SC7.C7_DESCRI) AS Descricao_Produto,
        COALESCE(SD1.D1_QUANT, SC7.C7_QUANT) AS Quantidade_Item,
        SC7.C7_TOTAL AS Valor_Total_Pedido,
        SD1.D1_TOTAL AS Valor_Total_Item_NF,
        RTRIM(SD1.D1_ITEM) AS Item_NF,
        RTRIM(SD1.D1_DOC) AS NF_Documento,
        RTRIM(SD1.D1_SERIE) AS Serie_NF,
        COALESCE(RTRIM(SD1.D1_FORNECE), RTRIM(SC7.C7_FORNECE)) AS Cod_Fornecedor,
        COALESCE(RTRIM(SD1.D1_LOJA), RTRIM(SC7.C7_LOJA)) AS Loja_Fornecedor,
        CASE WHEN ISDATE(SF1.F1_EMISSAO) = 1 THEN CAST(SF1.F1_EMISSAO AS DATE)
             WHEN ISDATE(SD1.D1_EMISSAO) = 1 THEN CAST(SD1.D1_EMISSAO AS DATE) END AS DT_Emissao_NF,
        CASE WHEN ISDATE(SF1.F1_RECBMTO) = 1 THEN CAST(SF1.F1_RECBMTO AS DATE) END AS Dt_Pre_Nota,
        CASE WHEN ISDATE(SF1.F1_DTDIGIT) = 1 THEN CAST(SF1.F1_DTDIGIT AS DATE)
             WHEN ISDATE(SD1.D1_DTDIGIT) = 1 THEN CAST(SD1.D1_DTDIGIT AS DATE) END AS DT_Escrituracao
    FROM SD1010 SD1 WITH (NOLOCK)
    FULL OUTER JOIN SC7010 SC7 WITH (NOLOCK)
        ON SC7.D_E_L_E_T_ = ''
       AND RTRIM(SD1.D1_FILIAL) = RTRIM(SC7.C7_FILIAL)
       AND RTRIM(SD1.D1_PEDIDO) = RTRIM(SC7.C7_NUM)
       AND RTRIM(SD1.D1_ITEMPC) = RTRIM(SC7.C7_ITEM)
       AND RTRIM(SD1.D1_FORNECE) = RTRIM(SC7.C7_FORNECE)
       AND RTRIM(SD1.D1_LOJA) = RTRIM(SC7.C7_LOJA)
    LEFT JOIN SF1010 SF1 WITH (NOLOCK)
        ON SF1.D_E_L_E_T_ = ''
       AND RTRIM(SF1.F1_FILIAL) = RTRIM(SD1.D1_FILIAL)
       AND RTRIM(SF1.F1_DOC) = RTRIM(SD1.D1_DOC)
       AND RTRIM(SF1.F1_SERIE) = RTRIM(SD1.D1_SERIE)
       AND RTRIM(SF1.F1_FORNECE) = RTRIM(SD1.D1_FORNECE)
       AND RTRIM(SF1.F1_LOJA) = RTRIM(SD1.D1_LOJA)
    LEFT JOIN SY1010 SY1 WITH (NOLOCK)
        ON SY1.D_E_L_E_T_ = ''
       AND RTRIM(SY1.Y1_USER) = RTRIM(SC7.C7_USER)
    WHERE (SD1.D_E_L_E_T_ = '' OR SD1.D_E_L_E_T_ IS NULL)
      AND (SC7.C7_EMISSAO >= '20260101' OR SF1.F1_EMISSAO >= '20260101' OR SD1.D1_DTDIGIT >= '20260101')
),
CTE_BASE_UNIFICADA AS (
    SELECT
        COALESCE(P.Filial_Origem, RTRIM(SE2.E2_FILIAL)) AS Filial,
        P.Numero_Pedido,
        P.Item_Pedido,
        P.Data_Emissao_Pedido,
        P.Condicao_Pagto_Pedido,
        P.Cod_Usuario_Comprador,
        COALESCE(NULLIF(P.Nome_Comprador, ''), 'COMPRADOR ' + ISNULL(P.Cod_Usuario_Comprador, 'N/I')) AS Nome_Comprador,
        P.Cod_Produto,
        P.Descricao_Produto,
        P.Quantidade_Item,
        P.Valor_Total_Pedido,
        P.Valor_Total_Item_NF,
        P.Item_NF,
        COALESCE(P.NF_Documento, RTRIM(SE2.E2_NUM)) AS NF_Documento,
        COALESCE(P.Serie_NF, RTRIM(SE2.E2_PREFIXO)) AS Serie_NF,
        COALESCE(P.Cod_Fornecedor, RTRIM(SE2.E2_FORNECE)) AS Cod_Fornecedor,
        COALESCE(P.Loja_Fornecedor, RTRIM(SE2.E2_LOJA)) AS Loja_Fornecedor,
        COALESCE(NULLIF(RTRIM(SA2.A2_NOME), ''), RTRIM(SE2.E2_NOMFOR)) AS Razao_Social_Fornecedor,
        COALESCE(NULLIF(RTRIM(SA2.A2_NREDUZ), ''), RTRIM(SE2.E2_NOMFOR), RTRIM(SA2.A2_NOME)) AS Nome_Fornecedor,
        RTRIM(SA2.A2_CGC) AS CNPJ_CPF_Fornecedor,
        COALESCE(P.DT_Emissao_NF, CASE WHEN ISDATE(SE2.E2_EMISSAO) = 1 THEN CAST(SE2.E2_EMISSAO AS DATE) END) AS DT_Emissao_NF,
        P.Dt_Pre_Nota,
        P.DT_Escrituracao,
        RTRIM(SE2.E2_NUM) AS Numero_Titulo,
        RTRIM(SE2.E2_PARCELA) AS Parcela,
        RTRIM(SE2.E2_TIPO) AS Tipo_Titulo,
        CASE WHEN ISDATE(SE2.E2_EMISSAO) = 1 THEN CAST(SE2.E2_EMISSAO AS DATE) END AS DT_Emissao_Titulo,
        CASE WHEN ISDATE(SE2.E2_VENCREA) = 1 THEN CAST(SE2.E2_VENCREA AS DATE) END AS Data_Real_Pgto,
        CASE WHEN ISDATE(SE2.E2_BAIXA) = 1 AND SE2.E2_BAIXA > '19010101' THEN CAST(SE2.E2_BAIXA AS DATE) END AS DT_Baixa_SE2,
        B.DT_Movimentacao_SE5,
        COALESCE(B.DT_Movimentacao_SE5, CASE WHEN ISDATE(SE2.E2_BAIXA) = 1 AND SE2.E2_BAIXA > '19010101' THEN CAST(SE2.E2_BAIXA AS DATE) END) AS Dt_Movimentacao,
        SE2.E2_VALOR AS Valor_Titulo_Original,
        SE2.E2_VALLIQ AS Valor_Liquido_Baixado,
        B.Valor_Movimentado_SE5 AS Valor_Pago_SE5,
        COALESCE(NULLIF(SE2.E2_VALLIQ, 0), B.Valor_Movimentado_SE5, SE2.E2_VALOR, 0) AS Valor_Movimentado,
        B.Motivo_Baixa_SE5,
        B.Tipo_Doc_SE5,
        ROW_NUMBER() OVER (
            PARTITION BY
                COALESCE(RTRIM(SE2.E2_FILIAL), P.Filial_Origem),
                COALESCE(RTRIM(SE2.E2_NUM), P.NF_Documento),
                ISNULL(RTRIM(SE2.E2_PARCELA), ''),
                COALESCE(RTRIM(SE2.E2_FORNECE), P.Cod_Fornecedor),
                COALESCE(RTRIM(SE2.E2_LOJA), P.Loja_Fornecedor)
            ORDER BY P.Item_NF ASC, P.Item_Pedido ASC
        ) AS RN_PARCELA
    FROM CTE_PEDIDOS_NF P
    FULL OUTER JOIN SE2010 SE2 WITH (NOLOCK)
        ON SE2.D_E_L_E_T_ = ''
       AND RTRIM(SE2.E2_FILIAL) = P.Filial_Origem
       AND RTRIM(SE2.E2_NUM) = P.NF_Documento
       AND RTRIM(SE2.E2_FORNECE) = P.Cod_Fornecedor
       AND RTRIM(SE2.E2_LOJA) = P.Loja_Fornecedor
    LEFT JOIN CTE_SE5_BAIXAS B
        ON B.Filial_Baixa = RTRIM(SE2.E2_FILIAL)
       AND B.Numero_Titulo = RTRIM(SE2.E2_NUM)
       AND B.Parcela_Titulo = RTRIM(SE2.E2_PARCELA)
       AND B.Cod_Fornecedor = RTRIM(SE2.E2_FORNECE)
       AND B.Loja_Fornecedor = RTRIM(SE2.E2_LOJA)
    LEFT JOIN SA2010 SA2 WITH (NOLOCK)
        ON SA2.D_E_L_E_T_ = ''
       AND RTRIM(SA2.A2_COD) = COALESCE(P.Cod_Fornecedor, RTRIM(SE2.E2_FORNECE))
       AND RTRIM(SA2.A2_LOJA) = COALESCE(P.Loja_Fornecedor, RTRIM(SE2.E2_LOJA))
    WHERE (SE2.E2_EMISSAO >= '20260101' OR P.DT_Emissao_NF >= '2026-01-01' OR P.Data_Emissao_Pedido >= '2026-01-01')
)
SELECT

    FORMAT(COALESCE(U.Dt_Movimentacao, U.Data_Real_Pgto, U.DT_Emissao_NF), 'yyyy-MM') AS [Mes_Baixa],
    CASE WHEN U.RN_PARCELA = 1 THEN U.Valor_Movimentado ELSE 0 END AS [Val Liq Baix],
    CASE WHEN U.RN_PARCELA = 1 THEN U.Valor_Movimentado ELSE 0 END AS [Vlr.Movim.],
    CASE WHEN U.RN_PARCELA = 1 THEN U.Valor_Movimentado ELSE 0 END AS [Valor],
    DATEDIFF(DAY, U.DT_Emissao_NF, U.Dt_Pre_Nota) AS [SLA_Pre_Nota],
    DATEDIFF(DAY, U.Dt_Pre_Nota, U.DT_Escrituracao) AS [SLA_Digitacao],
    DATEDIFF(DAY, COALESCE(U.DT_Emissao_Titulo, U.DT_Emissao_NF), U.Data_Real_Pgto) AS [Dias_Pgto],
    DATEDIFF(DAY, U.DT_Escrituracao, U.Data_Real_Pgto) AS [Margem_Dias_Financeiro],
    U.RN_PARCELA
FROM CTE_BASE_UNIFICADA U
ORDER BY
    COALESCE(U.Dt_Movimentacao, U.Data_Real_Pgto, U.DT_Emissao_NF) DESC,
    U.NF_Documento ASC,
    U.Parcela ASC;
